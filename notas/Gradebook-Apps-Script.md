# Registro de notas — Apps Script para la planilla nueva

Pegá el código completo de abajo como `Code.gs` en un proyecto Apps Script vinculado a la planilla. La URL `/exec` final se configura en `apps/notas/config.js`.

- Sheet ID: `1d9yL3ZhjdQIxRQEFVXltRA_QtjIZc6UrcO_1NyT1oA4`
- Pestañas iniciales: `Registros`, `Config`, `Usuarios`, `Instrucciones`
- El proyecto ya crea/valida el encabezado canónico de `Registros` en primera ejecución. Config/Usuarios ya están inicializadas.
- Scope requerido: Sheets. No uses acceso público; limitar web app a usuarios autorizados.
- No cargar datos reales hasta implementar autorización por materia/escuela/curso.

function authenticate_(username,password){const user=String(username||'').trim(),pass=String(password||'');if(!user||user.length>254||pass.length<1||pass.length>200)throw Error('INVALID_CREDENTIALS');const s=book_().getSheetByName(CONFIG.USERS_SHEET);if(!s)throw Error('USERS_SHEET_NOT_FOUND');const rows=s.getDataRange().getDisplayValues();if(!rows.length||rows[0][0]!=='usuario'||rows[0][1]!=='pass')throw Error('USERS_SHEET_INVALID');const match=rows.slice(1).find(r=>String(r[0]||'').trim().toLowerCase()===user.toLowerCase());if(!match||!constantEqual_(pass,String(match[1]||'')))throw Error('INVALID_CREDENTIALS');const now=Date.now(),payload={iat:now,exp:now+CONFIG.SESSION_SECONDS*1000,sub:user,tag:credentialTag_(user,pass),nonce:Utilities.getUuid()},token=base64_(JSON.stringify(payload));return{ok:true,sessionToken:token+'.'+sign_(token),expiresAt:new Date(payload.exp).toISOString(),username:user};}
function requireSession_(token){if(!token||typeof token!=='string')throw Error('UNAUTHORIZED');const p=token.split('.');if(p.length!==2||!constantEqual_(p[1],sign_(p[0])))throw Error('UNAUTHORIZED');let x;try{x=JSON.parse(Utilities.newBlob(Utilities.base64DecodeWebSafe(p[0])).getDataAsString());}catch(_){throw Error('UNAUTHORIZED');}const s=book_().getSheetByName(CONFIG.USERS_SHEET);if(!s)throw Error('USERS_SHEET_NOT_FOUND');const rows=s.getDataRange().getDisplayValues(),u=rows.slice(1).find(r=>String(r[0]||'').trim().toLowerCase()===String(x.sub||'').toLowerCase());if(!x.exp||Date.now()>x.exp||!u||x.tag!==credentialTag_(String(u[0]).trim(),String(u[1])))throw Error('SESSION_EXPIRED');}

function sign_(text){const props=PropertiesService.getScriptProperties();let key=props.getProperty('SESSION_SIGNING_KEY');if(!key){key=Utilities.getUuid()+Utilities.getUuid()+Utilities.getUuid();props.setProperty('SESSION_SIGNING_KEY',key);}return Utilities.base64EncodeWebSafe(Utilities.computeHmacSha256Signature(text,key)).replace(/=+$/,'');}
function base64_(text){return Utilities.base64EncodeWebSafe(Utilities.newBlob(text).getBytes()).replace(/=+$/,'');}
function constantEqual_(a,b){a=String(a);b=String(b);let d=a.length^b.length,n=Math.max(a.length,b.length);for(let i=0;i<n;i++)d|=(a.charCodeAt(i%Math.max(a.length,1))||0)^(b.charCodeAt(i%Math.max(b.length,1))||0);return d===0;}

function listRecords_(q){const limit=Math.min(Math.max(Number(q.limit)||100,1),setting_('maxPageSize',CONFIG.MAX_PAGE_SIZE,1,CONFIG.MAX_PAGE_SIZE)),cursor=String(q.cursor||''),sh=dataSheet_(),last=sh.getLastRow(),records=[],version=String(PropertiesService.getScriptProperties().getProperty('DATA_VERSION')||'0');if(q.version!==undefined&&String(q.version)!==version)throw Error('DATA_CHANGED');let start=2;if(cursor){const ids=last>1?sh.getRange(2,1,last-1,1).getDisplayValues().flat():[],i=ids.indexOf(cursor);if(i<0)throw Error('INVALID_CURSOR');start=i+3;}let end=start,next='';while(end<=last&&records.length<limit){const count=Math.min(500,last-end+1),rows=sh.getRange(end,1,count,CONFIG.HEADERS.length).getValues();for(const r of rows){end++;if(r[0]&&r[5]!=='DELETED'){records.push(unpack_(r));if(records.length===limit){next=r[0];break;}}}if(!rows.length)break;}return{ok:true,records,nextCursor:next,version};}
function unpack_(r){let data;try{data=JSON.parse(r[3]);}catch(_){throw Error('INVALID_STORED_RECORD');}return{recordId:r[0],type:r[1],id:r[2],data,version:Number(r[4]),status:r[5],updatedAt:r[6],createdAt:r[7]};}
function validateRecord_(x){if(!x||typeof x!=='object'||!/^[-\w:]{1,200}$/.test(String(x.recordId||''))||!/^[-\w:]{1,120}$/.test(String(x.id||''))||x.recordId!==`${x.type}::${x.id}`||!['school','course','division','subject','subgroup','student','membership','cutoff','scale','assessment','grade'].includes(x.type)||!x.data||typeof x.data!=='object'||Array.isArray(x.data))throw Error('INVALID_RECORD');const d=x.data;if(x.type==='student'&&(!String(d.studentId||'').trim()||String(d.studentId).length>80||/[\u0000-\u001f\u007f]/.test(String(d.studentId))||!String(d.fullName||'').trim()||String(d.fullName).length>120))throw Error('INVALID_RECORD');if(x.type==='grade'&&(!d.studentId||!d.assessmentId||!['graded','absent','not_submitted','pending','exempt'].includes(d.status)||d.value===undefined||String(d.observation||'').length>1000||!/^[0-9]{4}-[0-9]{2}-[0-9]{2}$/.test(String(d.gradeDate||''))))throw Error('INVALID_RECORD');if(x.type==='membership'&&(!d.studentId||!d.schoolId||!d.subjectId||!d.courseId||!d.divisionId))throw Error('INVALID_RECORD');if(x.type==='grade'&&(!d.studentId||!d.assessmentId||!['graded','absent','not_submitted','pending','exempt'].includes(d.status)||d.value===undefined))throw Error('INVALID_RECORD');const json=JSON.stringify(d);if(json.length>12000)throw Error('INVALID_RECORD');return{recordId:x.recordId,type:x.type,id:x.id,dataJson:json};}
function saveBatch_(b){const items=b.records;if(!Array.isArray(items)||!items.length||items.length>setting_('maxBatchSize',CONFIG.MAX_BATCH_SIZE,1,CONFIG.MAX_BATCH_SIZE))throw Map();rows.forEach(r=>{if(r[1]==='student'){try{const d=JSON.parse(r[3]);if(d.studentId)studentIds.set(String(d.studentId),r[0]);}catch(_){}}});for(const x of valid){if(x.value.type==='student'){const Error('DUPLICATE_STUDENT_ID');studentIds.set(studentId,x.value.recordId);}}const now=new Date().toISOString(),results=[],writes=[];for(const x of valid){const p=PropertiesService.getScriptProperties();p.setProperty('DATA_VERSION',String(Number(p.getProperty('DATA_VERSION')||0)+1));}return{ok:true,results};}finally{lock.releaseLock();}}
